    f.sale_id,
    c.customer_name,

    p.category,

    f.quantity,

JOIN DIM_CUSTOMER c
    ON f.customer_id = c.customer_id
JOIN DIM_PRODUCT p
    ON f.product_id = p.product_id
JOIN DIM_BRANCH b
    ON f.branch_id = b.branch_id
JOIN DIM_DATE d
    ON f.date_id = d.date_id
